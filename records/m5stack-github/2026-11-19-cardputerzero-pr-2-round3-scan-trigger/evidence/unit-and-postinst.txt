[Unit]
Description=BeaconHound Community 2.4GHz Spectrum Telemetry Daemon (%i)
After=network.target

[Service]
Type=simple
User=%i
RuntimeDirectory=beaconhound
RuntimeDirectoryMode=0700
ExecStart=/usr/bin/beaconhound-lite-daemon
Restart=on-failure
RestartSec=2
AmbientCapabilities=CAP_NET_ADMIN
CapabilityBoundingSet=CAP_NET_ADMIN
NoNewPrivileges=true
RestrictSUIDSGID=true
ProtectSystem=strict
PrivateTmp=true
ProtectHome=true

[Install]
WantedBy=multi-user.target

if [ "$1" = "remove" ] || [ "$1" = "deconfigure" ]; then
    HAS_SYSTEMD=0
    if command -v systemctl >/dev/null 2>&1 && [ -d /run/systemd/system ]; then
        HAS_SYSTEMD=1
    fi

    for bh_link in "${WANTS_DIR}/${TEMPLATE_PREFIX}"*.service; do
        [ -e "$bh_link" ] || [ -L "$bh_link" ] || continue
        bh_unit=$(basename -- "$bh_link")
        if [ "$HAS_SYSTEMD" = "1" ] && bh_is_safe_unit_name "$bh_unit"; then
            systemctl stop "$bh_unit" >/dev/null 2>&1 || true
            systemctl disable "$bh_unit" >/dev/null 2>&1 || true
        fi
        if [ -L "$bh_link" ]; then
            rm -f -- "$bh_link"
        fi
    done

    if [ "$HAS_SYSTEMD" = "1" ]; then
        bh_active_units=$(systemctl list-units --plain --type=service --all --no-legend --no-pager "${TEMPLATE_PREFIX}*.service" 2>/dev/null | awk '{
            for (i = 1; i <= NF; i++) {
                if ($i ~ /^beaconhound-lite-daemon@[a-zA-Z0-9._][a-zA-Z0-9._-]*\.service$/) {
                    print $i
                }
            }
        }' || true)
        printf '%s\n' "$bh_active_units" | while IFS= read -r bh_unit; do
            [ -n "$bh_unit" ] || continue
            if bh_is_safe_unit_name "$bh_unit"; then
                systemctl stop "$bh_unit" >/dev/null 2>&1 || true
                systemctl disable "$bh_unit" >/dev/null 2>&1 || true
            fi
        done

        systemctl stop "$LEGACY_SERVICE" >/dev/null 2>&1 || true
        systemctl disable "$LEGACY_SERVICE" >/dev/null 2>&1 || true
    fi

    if [ -L "${WANTS_DIR}/${LEGACY_SERVICE}" ]; then
        rm -f -- "${WANTS_DIR}/${LEGACY_SERVICE}"
    fi
fi

exit 0
