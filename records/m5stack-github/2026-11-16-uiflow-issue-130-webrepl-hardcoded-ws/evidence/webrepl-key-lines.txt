   181	static void webrepl_queue_input(const char *data, size_t len) {
   182	    webrepl_rx_received_total += len;
   183	    for (size_t i = 0; i < len; i++) {
   184	        uint8_t ch = (uint8_t)data[i];
   185	        if (ch == 0x03) {
   186	            webrepl_interrupt();
   187	            continue;
   188	        }
   189	        if (ch == '\n') {
   190	            ch = '\r';
   191	        }
   192	        webrepl_rx_push(ch);
   193	    }
   194	    webrepl_wake_main_task();
   195	}

   299	static void webrepl_process_text_message(const char *data, int len) {
   300	    if (data == NULL || len <= 0) {
   301	        return;
   302	    }
   303
   304	    webrepl_last_activity_us = esp_timer_get_time();
   305
   306	    if (webrepl_text_equals(data, len, "PONG")) {
   307	        return;
   308	    }
   309	    if (webrepl_text_equals(data, len, "PING")) {
   310	        webrepl_need_pong = true;
   311	        return;
   312	    }
   313
   314	    cJSON *root = cJSON_ParseWithLength(data, len);
   315	    if (root == NULL) {
   316	        ESP_LOGD(WEBREPL_TAG, "ignore non-json text len=%d", len);
   317	        return;
   318	    }
   319
   320	    cJSON *type = cJSON_GetObjectItem(root, "type");
   321	    if (cJSON_IsString(type) && strcmp(type->valuestring, "clientMessage") == 0) {
   322	        cJSON *payload = cJSON_GetObjectItem(root, "payload");
   323	        if (cJSON_IsString(payload) && payload->valuestring != NULL) {
   324	            webrepl_queue_input(payload->valuestring, strlen(payload->valuestring));
   325	        } else if (cJSON_IsNumber(payload)) {
   326	            char number_buf[32];
   327	            int n = snprintf(number_buf, sizeof(number_buf), "%g", payload->valuedouble);
   328	            if (n > 0) {
   329	                webrepl_queue_input(number_buf, (size_t)n);
   330	            }
   331	        }
   332	    }
   333
   334	    cJSON_Delete(root);
   335	}
