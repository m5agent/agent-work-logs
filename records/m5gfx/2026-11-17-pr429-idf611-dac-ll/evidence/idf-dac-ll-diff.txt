static inline void dac_ll_pad_set_data_source(dac_channel_t channel, dac_data_source_t source)
{
    if (source == DAC_DATA_SOURCE_DMA) { SENS.sar_dac_ctrl1.dac_dig_force = true; }
    else {
        bool cw_en = (source == DAC_DATA_SOURCE_COSINE);
        if (channel == DAC_CHAN_0) { SENS.sar_dac_ctrl2.dac_cw_en1 = cw_en; }
        else if (channel == DAC_CHAN_1) { SENS.sar_dac_ctrl2.dac_cw_en2 = cw_en; }
        SENS.sar_dac_ctrl1.dac_dig_force = false;
    }
}
