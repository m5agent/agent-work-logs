## 2. The op helper: i2c_bit_off(dev, reg, bits) = masked write value 0, mask bits
  constexpr op_t i2c_bit_on(std::uint8_t dev, std::uint16_t reg, std::uint8_t bits)
  {
    return i2c_masked8(dev, reg, bits, bits);
  }

  constexpr op_t i2c_bit_off(std::uint8_t dev, std::uint16_t reg, std::uint8_t bits)
  {
    return i2c_masked8(dev, reg, 0, bits);
  }

// 0x16 differs from the legacy sequence on purpose: the M5PM1 function
// registers use 2 bits per pin, so gpio2 is cleared with bits 5:4 (keep 0xCF),
// not bit 2 (keep 0xFB, which changed the function of gpio1).
static const legacy_write_t legacy_sticks3[] = {
  { 0x6E, 0x09, 0x00, 0x00 }, { 0x6E, 0x16, 0x00, 0xCF },
  { 0x6E, 0x10, 0x04, 0xFF }, { 0x6E, 0x13, 0x00, 0xFB },
