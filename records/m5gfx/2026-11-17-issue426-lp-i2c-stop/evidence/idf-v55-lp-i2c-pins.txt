===== gpio.c gpio_set_direction =====
  306: esp_err_t gpio_set_direction(gpio_num_t gpio_num, gpio_mode_t mode)
  307: {
  308:     GPIO_CHECK(GPIO_IS_VALID_GPIO(gpio_num), "GPIO number error", ESP_ERR_INVALID_ARG);
  309: 
  310:     if ((GPIO_IS_VALID_OUTPUT_GPIO(gpio_num) != true) && (mode & GPIO_MODE_DEF_OUTPUT)) {
  311:         ESP_LOGE(GPIO_TAG, "io_num=%d can only be input", gpio_num);
  312:         return ESP_ERR_INVALID_ARG;
  313:     }
  314: 
  315:     esp_err_t ret = ESP_OK;
  316: 
  317:     if (mode & GPIO_MODE_DEF_INPUT) {
  318:         gpio_input_enable(gpio_num);
  319:     } else {
  320:         gpio_input_disable(gpio_num);
  321:     }
  322: 
  323:     if (mode & GPIO_MODE_DEF_OUTPUT) {
  324:         gpio_output_enable(gpio_num);
  325:     } else {
  326:         gpio_output_disable(gpio_num);
  327:     }
  328: 
  329:     if (mode & GPIO_MODE_DEF_OD) {
  330:         gpio_od_enable(gpio_num);
  331:     } else {
  332:         gpio_od_disable(gpio_num);
  333:     }
  334: 
  335:     return ret;
  336: }
