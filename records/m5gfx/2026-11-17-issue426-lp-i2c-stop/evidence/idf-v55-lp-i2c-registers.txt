 761: /**
 762:  * @brief Clear I2C bus, when the slave is stuck in a deadlock and keeps pulling the bus low,
 763:  *        master can controls the SCL bus to generate 9 CLKs.
 764:  *
 765:  * Note: The master cannot detect if deadlock happens, but when the scl_st_to interrupt is generated, a deadlock may occur.
 766:  *
 767:  * @param  hw Beginning address of the peripheral registers
 768:  * @param  slave_pulses When I2C master is IDLE, the number of pulses will be sent out.
 769:  * @param  enable True to start the state machine, otherwise, false
 770:  *
 771:  * @return None
 772:  */
 773: static inline void i2c_ll_master_clr_bus(i2c_dev_t *hw, uint32_t slave_pulses, bool enable)
 774: {
 775:     hw->scl_sp_conf.scl_rst_slv_num = slave_pulses;
 776:     hw->scl_sp_conf.scl_rst_slv_en = enable;
 777:     hw->ctr.conf_upgate = 1;
 778:     // hardware will clear scl_rst_slv_en after sending SCL pulses,
 779:     // and we should set conf_upgate bit to synchronize register value after this function.
 780: }
 781:

 789: static inline bool i2c_ll_master_is_bus_clear_done(i2c_dev_t *hw)
